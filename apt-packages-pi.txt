# System (apt) dependencies needed on Raspberry Pi OS (64-bit) before `pip install -r
# requirements-pi.txt` will succeed and before the pipeline can run end-to-end.
# One package per line. Comments (#) and blank lines are stripped by scripts/setup_pi.sh before
# passing this to `apt install`, so keep this format if you edit it.
#
# Checked against actual imports/usage, not a generic list:
#   - espeak-ng deliberately NOT included: this pipeline does not use a phonemizer. French text is
#     normalized by embedded_tts's own regex rules (custom_regex_rules.csv / do_normalize_txt.pl,
#     the latter currently unused/dead code) and phonetic input is user-typed `{s y z i}` literal
#     phonemes, not auto-generated via espeak. g2p_en/pypinyin are imported at module load time in
#     FastSpeech2/synthesize.py but only instantiated for English/Mandarin text, never for French.

# venv + headers/toolchain: some pinned wheels above may not have aarch64 wheels for every version
# and will need to build from source.
python3-venv
python3-dev
build-essential

# BLAS backend for numpy/scipy/torch CPU math.
libopenblas-dev

# soundfile/librosa (pulled in transitively by noisereduce) need libsndfile at runtime.
libsndfile1

# pydub's playback path shells out to ffplay on non-Windows (see requirements-pi.txt notes on
# simpleaudio/sounddevice being Windows-only); ffmpeg also covers any non-wav format conversions.
ffmpeg

# aplay -l, used by scripts/setup_pi.sh's ALSA-default-device step (deploy/audio/asound.conf) to
# confirm the IQaudio DAC is actually present before pinning ALSA's default output to it. Usually
# already present on Raspberry Pi OS's base image, but not guaranteed -- listed explicitly rather
# than assumed, same convention as every other dependency in this file.
alsa-utils

# unzip the pretrained-weight archives downloaded by scripts/setup_pi.sh.
unzip

# only needed if running with --gui (Tkinter); see embedded_tts/README.md "Avec interface graphique".
python3-tk

# GUI mood-icon chip grid (chatterbox/synthesis/backends/fastspeech2_hifigan/backend.py's
# GST_TOKEN_ICONS) uses plain Unicode emoji glyphs -- real-hardware check (fc-list on a Pi5
# kiosk image) found no emoji-capable font installed at all (only dejavu/noto-mono, neither of
# which cover the Emoji block), so every one of those 12 icons rendered as a blank box. This is a
# cosmetic-only dependency (nothing crashes without it, the chips just show a blank glyph instead
# of a face), not a hard requirement -- everything else in this repo (the Texte keyboard's
# play/stop button, the battery indicator) was deliberately moved OFF font-glyph icons entirely
# instead of depending on a package like this one (see docs/context/CHANGELOG.md, "real-hardware
# feedback round 2").
fonts-noto-color-emoji

# i2cdetect, used to verify the optional INA226 amp-branch power sensor is wired correctly at
# i2c-1 @ 0x40 (and doesn't collide with the IQaudio DAC at 0x4c) before a --benchmark --play run;
# see embedded_tts/README.md "Profilage" / "Capteur INA226".
i2c-tools

# gpiozero's lgpio backend (physical switches) has no prebuilt PyPI wheel for aarch64 on newer
# Python builds (observed on Debian 13/trixie's Python 3.13) -- pip falls back to compiling it
# from source, which needs swig to generate the C extension bindings, and liblgpio-dev to link
# against (liblgpio1, the runtime .so, ships by default on Raspberry Pi OS trixie, but not the
# -dev headers/linkable lib pip's build needs). Neither is needed on Debian 12/bookworm's Python
# 3.11, where a prebuilt wheel exists.
swig
liblgpio-dev

# Kiosk display: plain Xorg (deploy/xorg-kiosk/, docs/kiosk/KIOSK.md), the CURRENT default --
# NOT cage/Wayland. Real Pi5 hardware bring-up (2026-07-31, docs/context/CHANGELOG.md) found a
# reproducible SIGSEGV deep inside libwlroots (Raspberry Pi Foundation's own build), triggered by
# essentially any input event, with no fixed package available in either the Raspberry Pi or
# Debian repos -- see deploy/xorg-kiosk/README.md for the full writeup and how to revert if a fix
# ever lands. xinit provides `startx`; x11-xserver-utils provides `xrandr` (bring-up debugging
# only, not required at runtime).
xserver-xorg
xinit
x11-xserver-utils

# Legacy/reference only, NOT installed by scripts/setup_pi.sh by default -- the originally
# finalized compositor choice (docs/kiosk/KIOSK.md), abandoned per the wlroots bug above. Left
# commented here (rather than deleted) so reverting is a one-line uncomment + `sudo apt install`
# if a fixed libwlroots package ever reaches the repos; deploy/systemd/chatterbox-gui.service
# (cage's own unit) is untouched and ready to re-enable alongside it.
# cage
# xwayland
